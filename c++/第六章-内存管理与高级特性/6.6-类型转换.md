# 6.6 类型转换

C++ 提供四种显式类型转换运算符。static_cast 用于编译时已知的安全转换。dynamic_cast 用于运行时多态类型转换，通过 RTTI 检查类型安全性。const_cast 用于添加或移除 const 属性。reinterpret_cast 用于底层的位模式重新解释，是最危险的转换。

```cpp
#include <iostream>
using namespace std;
class Base {
public:
virtual ~Base() {}
virtual void show() { cout << "Base" << endl; }
};
class Derived : public Base {
public:
void show() override { cout << "Derived" << endl; }
};
int main() {
// static_cast
double d = 3.14;
int i = static_cast<int>(d);
cout << "double转int: " << i << endl;
// dynamic_cast
Base* base = new Derived();
Derived* derived = dynamic_cast<Derived*>(base);
if (derived) derived->show();
delete base;
// const_cast
const int cval = 10;
int& ref = const_cast<int&>(cval);
// reinterpret_cast
int num = 0x41424344;
char* chars = reinterpret_cast<char*>(&num);
cout << "字节: " << chars[0] << chars[1] << endl;
return 0;
}
```
