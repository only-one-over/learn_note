# 1.3 运算符

运算符是对操作数执行特定操作的符号。C++ 提供了丰富的运算符：算术运算符（+、-、*、/、%）用于数学计算；关系运算符（==、!=、<、>）用于比较大小并返回布尔值；逻辑运算符（&&、||、!）用于布尔逻辑组合；位运算符（&、|、^、~、<<、>>）直接操作整数的二进制位。

运算符优先级决定了表达式中各运算的执行顺序。一般规则是：算术运算符 > 关系运算符 > 逻辑运算符。当不确定优先级时，建议使用括号明确意图。自增（++）和自减（--）分前置和后置两种形式：前置先修改再使用，后置先使用再修改。位运算在底层编程中非常重要。

```cpp
#include <iostream>
using namespace std;
int main() {
// 算术运算符
int a = 10, b = 3;
cout << "a + b = " << a + b << endl;  // 13
cout << "a / b = " << a / b << endl;  // 3 (整数除法截断)
cout << "a % b = " << a % b << endl;  // 1
// 自增自减
int c = 5;
cout << "c++ = " << c++ << endl;  // 5 (后置: 先用后加)
cout << "c = " << c << endl;      // 6
cout << "++c = " << ++c << endl;  // 7 (前置: 先加后用)
// 关系与逻辑运算符
bool result = (a > 5) && (b < 10); // true && true = true
cout << "result = " << result << endl;
// 位运算
unsigned int x = 5;  // 0101
unsigned int y = 3;  // 0011
cout << "x & y = " << (x & y) << endl;   // 0001 = 1
cout << "x | y = " << (x | y) << endl;   // 0111 = 7
cout << "x << 1 = " << (x << 1) << endl; // 1010 = 10
return 0;
}
```
